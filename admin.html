// ========================================== // 1. CONFIGURACIÓN DE CREDENCIALES CORREGIDA // ========================================== const URL_SUPABASE = "https://supabase.co"; const KEY_SUPABASE = "sb_publishable_udTmjDidXlwFaOwEpeEKug_u4BtdXeB"; // Usamos 'window.supabase' para evitar que se pisen los nombres en el navegador const clienteSupabase = window.supabase.createClient(URL_SUPABASE, KEY_SUPABASE, { realtime: { params: { eventsPerSecond: 10 }, timeout: 40000 } }); // ========================================== // 2. CONFIGURACIÓN DE AUDIO (SIRENA) // ========================================== let audioSirena = new Audio('https://mixkit.co'); audioSirena.loop = true; let audioHabilitado = false; function inicializarSistemaAudio() { audioSirena.play().then(() => { audioSirena.pause(); audioSirena.currentTime = 0; audioHabilitado = true; document.getElementById('btnIniciar').style.backgroundColor = "var(--success)"; document.getElementById('btnIniciar').innerHTML = "🟢 SIRENA CONFIGURADA Y LISTA"; setTimeout(() => { document.getElementById('btnIniciar').style.display = "none"; }, 1500); console.log("Audio aprobado con éxito."); }).catch(error => { console.error("Fallo al inicializar el audio:", error); alert("Por favor, haz clic nuevamente en el botón para activar el sonido."); }); } function sonarSirena() { if (audioHabilitado) { audioSirena.play().catch(e => console.log("La sirena no pudo sonar:", e)); } else { console.log("🚨 Audio no habilitado. Presioná el botón rojo."); } } function detenerSirena() { audioSirena.pause(); audioSirena.currentTime = 0; } // ========================================== // 3. CONTROL DE ALERTAS Y FLUJO DE DATOS // ========================================== let idAlertaActiva = null; function desplegarAlertaEnPantalla(alerta) { idAlertaActiva = alerta.id; document.getElementById('alerta-id').innerText = alerta.id || 'N/A'; document.getElementById('alerta-usuario').innerText = alerta.usuario || alerta.emisor_name || 'Usuario Anónimo'; document.getElementById('alerta-detalle').innerText = alerta.detalle || alerta.mensaje || 'Señal de Pánico Activada'; document.getElementById('alerta-hora').innerText = new Date(alerta.created_at || Date.now()).toLocaleTimeString(); document.getElementById('sin-alertas').style.display = 'none'; document.getElementById('contenedor-alerta').style.display = 'block'; sonarSirena(); } function atenderAlertaActual() { if (!idAlertaActiva) return; detenerSirena(); limpiarPantallaAlerta(); alert("Alerta marcada como ATENDIDA."); } function cerrarAlertaActual() { if (!idAlertaActiva) return; detenerSirena(); limpiarPantallaAlerta(); } function limpiarPantallaAlerta() { idAlertaActiva = null; document.getElementById('contenedor-alerta').style.display = 'none'; document.getElementById('sin-alertas').style.display = 'block'; } // ========================================== // 4. CONEXIÓN CORREGIDA USANDO EL NUEVO CLIENTE // ========================================== let canalAlertas = null; function conectarTiempoReal() { const statusBadge = document.getElementById('network-status'); canalAlertas = clienteSupabase .channel('monitoreo-alertas-canal') .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'alertas' }, (payload) => { console.log("🚨 Nueva alerta en tiempo real:", payload.new); desplegarAlertaEnPantalla(payload.new); }) .subscribe((status, err) => { console.log("Estado del canal Supabase:", status); if (status === 'SUBSCRIBED') { statusBadge.innerText = "Conectado"; statusBadge.className = "status-badge status-connected"; } if (status === 'CHANNEL_ERROR' || status === 'CLOSED') { statusBadge.innerText = "Error de Red (Reconectando...)"; statusBadge.className = "status-badge status-disconnected"; if(canalAlertas) { clienteSupabase.removeChannel(canalAlertas); } setTimeout(() => { conectarTiempoReal(); }, 3000); } }); } window.onload = () => { conectarTiempoReal(); };
